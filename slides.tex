\section{Start here}

\begin{frame}{How to use this file}
  \begin{enumerate}
    \item Browse this PDF to find the layout you want.
    \item Search for that slide title in \texttt{slides.tex}.
    \item Copy its complete \texttt{frame}, then replace the content.
    \item Delete the tutorial slides and compile your presentation.
  \end{enumerate}
  \takeaway{One source file contains the guide and every example.}
  \source{Build locally with \texttt{make} or run pdfLaTeX twice.\newline
    In Overleaf, select \texttt{slides.tex} and the pdfLaTeX compiler.}
\end{frame}

  \begin{columns}[T,onlytextwidth]

      The four fields above the document set the cover text.
      \medskip

\begin{frame}[fragile]{Speaker notes}
  Add a note inside any frame, or immediately after \verb|\maketitle|:
  \medskip
  {\small
\begin{verbatim}
\note{Explain why this choice matters to the audience.}
\end{verbatim}
  }
  \medskip
  Notes are hidden by default. To include them, add these lines before
  \verb|\begin{document}|:
  {\small
\begin{verbatim}
\setbeameroption{show notes}
\setbeamertemplate{note page}[plain]
\end{verbatim}
  }
  \source{Use \texttt{show only notes} for a notes-only PDF, or remove the
    option to return to the normal presentation.}
  \note{This is an actual speaker note. Enable show notes near the top of the file to see it.}
\end{frame}

    \end{column}
    \begin{column}{0.50\textwidth}
      \small
      Burgundy is the default for blocks, the cover, headings, and takeaways:
\begin{verbatim}
\definecolor{burgundy}
  {HTML}{7A003C}
\end{verbatim}
      Use six hexadecimal digits without a hash. A coloured block keeps
      the same typography and spacing for every research role.
      \medskip

      \texttt{\textbackslash alert\{words\}} provides
      \alert{burgundy emphasis} within a sentence.
    \end{column}
  \end{columns}
  \source{Original burgundy; other accents: \href{https://www.concordia.ca/web/design/ui-kit-style-guide-accessibility/web-palette-new.html}{Concordia University web colour palette}.
    Research categories are this template's convention.}
\end{frame}

\begin{frame}[fragile]{Everyday fixes}
  \small
  \begin{description}
    \item[Literal characters] Write \verb|\%|, \verb|\&|, \verb|\_|,
      \verb|\#|, and \verb|\$| in ordinary text.
    \item[Code on a slide] Use \verb|\begin{frame}[fragile]{Title}|.
    \item[Long section name] Use \verb|\section[Short]{Full name}|.
    \item[Missing totals or citations] Compile twice; \texttt{make} does this.
    \item[Text or figures overflow] Shorten, split the slide, or reduce figure height.
    \item[Compilation fails] Start with the first error in \texttt{slides.log}.
  \end{description}
  \takeaway{To create your own deck, keep the preamble and the frames you need.
    Replace this guide's content and keep both style files beside \texttt{slides.tex}.}
\end{frame}
