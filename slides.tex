\section{Start here}

\begin{frame}{How to use this file}
  \begin{enumerate}
    \item Browse this PDF to find the layout you want.
    \item Search for that slide title in \texttt{slides.tex}.
    \item Copy its complete \texttt{frame}, then replace the content.
    \item Delete the tutorial slides and compile your presentation.
  \end{enumerate}
  \takeaway{One source file contains the guide and every example.}
  \source{Build locally with \texttt{make} or the VS Code build task.\newline
    In Overleaf, select \texttt{slides.tex} and the pdfLaTeX compiler.}
\end{frame}

\begin{frame}[fragile]{Everyday fixes}
  \small
  \begin{description}
    \item[Literal characters] Write \verb|\%|, \verb|\&|, \verb|\_|,
      \verb|\#|, and \verb|\$| in ordinary text.
    \item[Code on a slide] Use \verb|\begin{frame}[fragile]{Title}|.
    \item[Long section name] Use \verb|\section[Short]{Full name}|.
    \item[Missing totals or citations] Compile twice; \texttt{make} does this.
    \item[Text or figures overflow] Shorten, split the slide, or reduce figure height.
    \item[Compilation fails] Start with the first error in \texttt{slides.log}.
  \end{description}
  \takeaway{To create your own deck, keep the preamble and the frames you need.
    Replace this guide's content and keep both style files beside \texttt{slides.tex}.}
\end{frame}
